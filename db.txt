-- ============================================================
--  HONKAI STAR RETAIL — Database Schema
--  MySQL 8.x  |  XAMPP 8.2.12
-- ============================================================

CREATE DATABASE IF NOT EXISTS honkai_star_retail
  CHARACTER SET utf8mb4
  COLLATE utf8mb4_unicode_ci;

USE honkai_star_retail;

-- ============================================================
--  TABLE: users
--  role: 'admin' | 'user'
--  Admin hanya bisa dibuat manual (INSERT langsung / Postman)
--  User terbuat otomatis saat register email atau OAuth Google
-- ============================================================
CREATE TABLE users (
  id            INT UNSIGNED    NOT NULL AUTO_INCREMENT,
  name          VARCHAR(100)    NOT NULL,
  email         VARCHAR(150)    NOT NULL UNIQUE,
  password      VARCHAR(255)        NULL COMMENT 'NULL jika login via Google OAuth',
  google_id     VARCHAR(100)        NULL UNIQUE COMMENT 'Diisi jika login via Google',
  role          ENUM('admin','user') NOT NULL DEFAULT 'user',
  avatar_url    VARCHAR(500)        NULL,
  wallet        DECIMAL(15,2)   NOT NULL DEFAULT 0.00 COMMENT 'Saldo virtual dalam Credits',
  created_at    TIMESTAMP       NOT NULL DEFAULT CURRENT_TIMESTAMP,
  updated_at    TIMESTAMP       NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,

  PRIMARY KEY (id),
  INDEX idx_email (email),
  INDEX idx_google_id (google_id)
) ENGINE=InnoDB;


-- ============================================================
--  TABLE: products
--  Dikelola bersama oleh semua admin (shared data)
--  image_url pakai link URL (tidak upload file)
-- ============================================================
CREATE TABLE products (
  id            INT UNSIGNED    NOT NULL AUTO_INCREMENT,
  name          VARCHAR(150)    NOT NULL,
  type          VARCHAR(100)    NOT NULL COMMENT 'Contoh: Light Cone, Relic, Material, Credit',
  description   TEXT                NULL,
  stock         INT UNSIGNED    NOT NULL DEFAULT 0,
  image_url     VARCHAR(500)        NULL,
  price         DECIMAL(15,2)   NOT NULL DEFAULT 0.00 COMMENT 'Harga dalam Credits',
  created_by    INT UNSIGNED        NULL COMMENT 'FK ke users (admin yang buat)',
  created_at    TIMESTAMP       NOT NULL DEFAULT CURRENT_TIMESTAMP,
  updated_at    TIMESTAMP       NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,

  PRIMARY KEY (id),
  INDEX idx_type (type),
  CONSTRAINT fk_product_creator
    FOREIGN KEY (created_by) REFERENCES users(id)
    ON DELETE SET NULL
) ENGINE=InnoDB;


-- ============================================================
--  TABLE: cart
--  Keranjang belanja user (sebelum checkout)
--  Satu user + satu produk = satu baris (quantity bisa > 1)
-- ============================================================
CREATE TABLE cart (
  id            INT UNSIGNED    NOT NULL AUTO_INCREMENT,
  user_id       INT UNSIGNED    NOT NULL,
  product_id    INT UNSIGNED    NOT NULL,
  quantity      INT UNSIGNED    NOT NULL DEFAULT 1,
  added_at      TIMESTAMP       NOT NULL DEFAULT CURRENT_TIMESTAMP,
  updated_at    TIMESTAMP       NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,

  PRIMARY KEY (id),
  UNIQUE KEY uq_cart_user_product (user_id, product_id),
  CONSTRAINT fk_cart_user
    FOREIGN KEY (user_id) REFERENCES users(id)
    ON DELETE CASCADE,
  CONSTRAINT fk_cart_product
    FOREIGN KEY (product_id) REFERENCES products(id)
    ON DELETE CASCADE
) ENGINE=InnoDB;


-- ============================================================
--  TABLE: inventory
--  Produk yang sudah dibeli user (Buy Now atau Checkout Cart)
--  Satu user + satu produk = satu baris, quantity bisa bertambah
-- ============================================================
CREATE TABLE inventory (
  id            INT UNSIGNED    NOT NULL AUTO_INCREMENT,
  user_id       INT UNSIGNED    NOT NULL,
  product_id    INT UNSIGNED    NOT NULL,
  quantity      INT UNSIGNED    NOT NULL DEFAULT 1,
  bought_at     TIMESTAMP       NOT NULL DEFAULT CURRENT_TIMESTAMP,
  updated_at    TIMESTAMP       NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,

  PRIMARY KEY (id),
  UNIQUE KEY uq_inventory_user_product (user_id, product_id),
  CONSTRAINT fk_inventory_user
    FOREIGN KEY (user_id) REFERENCES users(id)
    ON DELETE CASCADE,
  CONSTRAINT fk_inventory_product
    FOREIGN KEY (product_id) REFERENCES products(id)
    ON DELETE CASCADE
) ENGINE=InnoDB;


-- ============================================================
--  TABLE: wallet_transactions
--  Riwayat semua perubahan saldo (topup & pembelian)
--  type: 'topup' | 'purchase'
-- ============================================================
CREATE TABLE wallet_transactions (
  id            INT UNSIGNED    NOT NULL AUTO_INCREMENT,
  user_id       INT UNSIGNED    NOT NULL,
  type          ENUM('topup','purchase') NOT NULL,
  amount        DECIMAL(15,2)   NOT NULL COMMENT 'Positif = topup, negatif = purchase',
  description   VARCHAR(255)        NULL COMMENT 'Keterangan transaksi',
  created_at    TIMESTAMP       NOT NULL DEFAULT CURRENT_TIMESTAMP,

  PRIMARY KEY (id),
  INDEX idx_wallet_user (user_id),
  CONSTRAINT fk_wallet_user
    FOREIGN KEY (user_id) REFERENCES users(id)
    ON DELETE CASCADE
) ENGINE=InnoDB;
